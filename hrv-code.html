HRV clustering: R code

Cluster analyses using k-means

# 1. Choose HRV clustering variables
# Baseline SDNN, HF/LF, LF, and HF

HRV_cluster_vars <- HRV_Data[, c(
  "sdnbasT1", "hfrpbaT1", "lfbaT1", "lhratbaT1"
)]

# Remove participants with missing data
HRV_cluster_vars <- na.omit(HRV_cluster_vars)

# Observe ranges
summary(HRV_cluster_vars)

# 2. Standardize HRV variables
Scaled_HRV <- scale(HRV_cluster_vars)

# 3. Check distributions
hist(Scaled_HRV)

# 4. Examine the number of clusters with the elbow method
set.seed(240)

elbow_plot <- factoextra::fviz_nbclust(
  Scaled_HRV,
  stats::kmeans,
  method = "wss",
  nstart = 50
)

elbow_df <- data.frame(
  k = as.integer(as.character(elbow_plot$data$clusters)),
  WSS = elbow_plot$data$y
)

elbow_df

# 5. Run k-means cluster analyses
set.seed(240)
kmeans3 <- stats::kmeans(Scaled_HRV, centers = 3, nstart = 50)

set.seed(240)
kmeans4 <- stats::kmeans(Scaled_HRV, centers = 4, nstart = 50)

# 6. Add cluster assignments to the analysis data
final_data <- HRV_cluster_vars
final_data$cluster <- kmeans4$cluster

final_data$cluster_1 <- as.integer(final_data$cluster == 1)
final_data$cluster_2 <- as.integer(final_data$cluster == 2)
final_data$cluster_3 <- as.integer(final_data$cluster == 3)
final_data$cluster_4 <- as.integer(final_data$cluster == 4)

table(final_data$cluster)

head(final_data[, c(
  "cluster", "cluster_1", "cluster_2", "cluster_3", "cluster_4"
)])

# 7. Visualize the clusters
factoextra::fviz_cluster(
  kmeans4,
  data = Scaled_HRV,
  main = "K-Means Clusters"
)

# 8. Export
haven::write_sav(final_data, "final_data.sav")
